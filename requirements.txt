# THE EARS - offline speech recognition (the microphone -> words)
vosk

# MICROPHONE - opens your sound device
sounddevice

# THE VOICE - talks to the voice built into Windows
# (this package provides win32com and win32api)
pywin32

# INTERNET - used for Gemini, Ollama, the weather and the online check
requests

# LAPTOP INFO - battery level, memory, CPU
psutil

# THE CLOUD BRAIN - Google Gemini
# Optional. The app still runs without it, using Ollama only.
google-genai
